# Arquivo: A.yml
title: Remote Thread Creation from PowerShell Process
id: f7a8b9c0-d1e2-3456-fabc-456789012345
status: experimental
description: Detects remote thread creation originating from a PowerShell process into another process, a code injection technique used to execute shellcode or migrate into other processes while evading detection.
references:
    - https://speakerdeck.com/heirhabarov/hunting-for-powershell-abuse
author: Senior Threat Detection Engineer
date: 2024-01-01
tags:
    - attack.execution
    - attack.privilege_escalation
    - attack.defense_evasion
    - attack.t1055
    - attack.t1059.001
logsource:
    product: windows
    category: create_remote_thread
detection:
    selection:
        SourceImage|endswith:
            - '\powershell.exe'
            - '\pwsh.exe'
    filter_main_telemetry:
        SourceParentImage|endswith:
            - '\CompatTelRunner.exe'
    condition: selection and not 1 of filter_main_*
falsepositives:
    - Legitimate administrative PowerShell scripts that interact with other processes in uncommon ways
    - Security tooling or monitoring agents that use PowerShell for process inspection
level: high

# Arquivo: C.yml
title: Remote Thread Creation From PowerShell Process
id: 5a8d3f2b-9c4e-4b7a-8d1f-2e6c7a9b0f3d
status: experimental
description: Detects CreateRemoteThread events where the source process is PowerShell (Windows PowerShell or PowerShell Core). Remote thread creation from PowerShell is a common code injection technique used to execute shellcode or migrate into other processes for stealthy execution. Excludes known benign cases where PowerShell is spawned by the Windows compatibility telemetry runner.
author: Senior Threat Detection Engineer
date: 2026/06/07
references:
  - [https://speakerdeck.com/heirhabarov/hunting-for-powershell-abuse](https://speakerdeck.com/heirhabarov/hunting-for-powershell-abuse)
tags:
  - attack.execution
  - attack.persistence
  - attack.t1055
  - windows
  - sysmon
logsource:
  product: windows
  category: create_remote_thread
  description: Sysmon CreateRemoteThread events must be enabled and collected (Event ID 8). Ensure process image and parent image fields are captured.
detection:
  selection:
    EventID: 8
    SourceImage|endswith:
      - '\\Windows\\System32\\WindowsPowerShell\\v1.0\\powershell.exe'
      - '\\System32\\WindowsPowerShell\\v1.0\\powershell.exe'
      - '\\powershell.exe'
      - '\\PowerShell.exe'
      - '\\pwsh.exe'
      - '\\Program Files\\PowerShell\\7\\pwsh.exe'
      - '\\Program Files (x86)\\PowerShell\\7\\pwsh.exe'
  filter_main_compattelemetry:
    ParentImage|endswith:
      - '\\CompatTelRunner.exe'
      - '\\CompatTelRunner.exe'
  condition: selection and not 1 of filter_main_compattelemetry
falsepositives:
  - Legitimate administrative automation or remote management workflows that use PowerShell to inject into helper processes.
  - Security tooling or EDR components that perform process injection for remediation or monitoring.
level: high
